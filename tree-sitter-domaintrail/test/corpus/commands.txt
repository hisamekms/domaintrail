context A {
  aggregate Order {
    command cancel(reason: Reason) "取り消す"
      Draft | Confirmed -> Cancelled
      requires noteForOther "備考" {
        !(reason == Reason.Other) || note != null
      } else NoteRequired
      requires documented "説明のみ" else Common.Forbidden
      changes note, reason
      emits Cancelled, Refund? when Confirmed ->, Audit[0..*] when -> Cancelled | Closed
      errors Sales.Locked, Other
  }
}

context A {
  aggregate Rota {
    create command open(id: RotaId) {
      outcome opened: -> Draft
        changes id = id, revision = 0, shifts = []
    }
    command addShift(shift: Shift) {
      outcome added: Draft -> Draft
        changes shifts = [...before.shifts, shift], revision
        ensures { after.revision > before.revision }
    }
    command touch()
      Draft -> Draft
      changes revision = before.revision + 1
      ensures { after.revision > before.revision }
  }
}
