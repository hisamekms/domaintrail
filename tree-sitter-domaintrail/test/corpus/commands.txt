Transitions, requirements, changes, emits and errors
================================================================================

context A {
  aggregate Order {
    command cancel(reason: Reason) "取り消す"
      Draft | Confirmed -> Cancelled
      requires noteForOther "備考" {
        reason == Reason.Other implies exists(note)
      } else NoteRequired
      requires documented "説明のみ" else Common.Forbidden
      changes note, reason
      emits Cancelled, Refund? when Confirmed ->, Audit[0..*] when -> Cancelled | Closed
      errors Sales.Locked, Other
  }
}
